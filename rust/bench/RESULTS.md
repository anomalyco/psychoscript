## Commands

```sh
bash bench/compare.sh  # reference at the repository root, or PSYCHOSCRIPT_REF=<package dir>
cargo build --release --locked -p psychoscript-bench --bin psychoscript-bench --example embed
# (a)
bun bench/ref/bench.ts --iterations 11 <slug>...
target/release/psychoscript-bench --filter <slug> --iterations 11 --json
# (b) 20 timed runs each after one untimed run, wall time from bash $EPOCHREALTIME (perl's clock without it)
bun bench/ref/embed.ts
target/release/examples/embed
# (c) -l instead of -v on macOS
/usr/bin/time -v bun bench/ref/run-case.ts <slug>
/usr/bin/time -v target/release/psychoscript-bench --filter <slug> --iterations 0
# (d) stat -f %z on macOS
strip -o embed target/release/examples/embed && stat -c %s embed
stat -c %s "$(readlink -f "$(command -v bun)")"
# (e) values must match their originals, on both sides
bun bench/ref/variants.ts --iterations 11 <slug>...
target/release/psychoscript-bench --cases bench/variants --filter <slug> --iterations 11 --json
# (f)
bun bench/ref/plugin.ts --runs 2000 bench/plugin/reducer.js bench/plugin/reducer-padded.js bench/plugin/reducer-100k.js
target/release/psychoscript-bench --plugin --runs 2000 bench/plugin/reducer.js bench/plugin/reducer-padded.js bench/plugin/reducer-100k.js
```
